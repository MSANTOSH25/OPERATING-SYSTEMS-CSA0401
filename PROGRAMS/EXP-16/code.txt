#include <stdio.h>

struct Employee
{
    int id;
    char name[30];
    float salary;
};

int main()
{
    struct Employee emp[50];
    int n, position;

    printf("Enter number of employees: ");
    scanf("%d", &n);

    if (n <= 0 || n > 50)
    {
        printf("Invalid number of employees.\n");
        return 0;
    }

    /* Store employee records */
    for (int i = 0; i < n; i++)
    {
        printf("\nEnter details of Employee %d\n", i + 1);

        printf("Enter Employee ID: ");
        scanf("%d", &emp[i].id);

        printf("Enter Employee Name: ");
        scanf("%29s", emp[i].name);

        printf("Enter Salary: ");
        scanf("%f", &emp[i].salary);
    }

    /*
       Random access:
       Directly access the required employee record
       without reading the previous records.
    */
    printf("\nEnter employee position to access randomly: ");
    scanf("%d", &position);

    if (position < 1 || position > n)
    {
        printf("Invalid employee position.\n");
        return 0;
    }

    printf("\n--- Employee Details ---\n");
    printf("Employee ID : %d\n", emp[position - 1].id);
    printf("Name        : %s\n", emp[position - 1].name);
    printf("Salary      : %.2f\n", emp[position - 1].salary);

    return 0;
}
